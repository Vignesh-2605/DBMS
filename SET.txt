import pandas as pd

def main():
    d = pd.read_csv('disease.csv')
    h = ['0'] * (len(d.columns) - 1)

    for i in range(len(d)):
        if d.iloc[i, -1].lower() == 'positive':
            for j in range(len(h)):
                if h[j] == '0':
                    h[j] = d.iloc[i, j]
                elif h[j] != d.iloc[i, j]:
                    h[j] = '?'

    print("Most specific hypothesis:", h)

main()

✅ Set 2: Candidate Elimination Algorithm
🔹 Q1: Determine possible rules for job shortlisting using resume data
python
Copy
Edit
import pandas as pd

def consistent(h, ex):
    for x, y in zip(h, ex):
        if x != '?' and x != y:
            return False
    return True

d = pd.read_csv('resumes.csv')
s = ['0'] * (len(d.columns) - 1)
g = [['?'] * (len(d.columns) - 1)]

for i in range(len(d)):
    row = d.iloc[i]
    if row[-1] == 'Yes':
        for j in range(len(s)):
            if s[j] == '0':
                s[j] = row[j]
            elif s[j] != row[j]:
                s[j] = '?'
        g = [h for h in g if consistent(h, row[:-1])]
    else:
        g_new = []
        for h in g:
            for j in range(len(h)):
                if h[j] == '?':
                    h1 = h.copy()
                    h1[j] = s[j]
                    g_new.append(h1)
        g = g_new

print("Specific Hypothesis:", s)
print("General Hypotheses:", g)

python
Copy
Edit
import pandas as pd
from sklearn.tree import DecisionTreeClassifier, export_text
from sklearn.preprocessing import LabelEncoder

d = pd.read_csv("weather.csv")
le = LabelEncoder()
for col in d.columns:
    d[col] = le.fit_transform(d[col])

x = d.iloc[:, :-1]
y = d.iloc[:, -1]
model = DecisionTreeClassifier(criterion="entropy")
model.fit(x, y)
print(export_text(model, feature_names=list(x.columns)))
